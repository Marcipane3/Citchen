      let pickerLoaded = null;

      export function openPickerForFile(mimeType, onFilePicked) {
        if (!TOKEN) return;  // Not signed in — silently abort
        if (!pickerLoaded) {
          pickerLoaded = new Promise((resolve) => gapi.load("picker", resolve));
        }
        pickerLoaded.then(() => {
          const view = new google.picker.View(google.picker.ViewId.DOCS);
          view.setMimeTypes(mimeType);
          const builder = new google.picker.PickerBuilder()
            .setOAuthToken(TOKEN)
            .addView(view)
            .setCallback((data) => {
              if (data.action === google.picker.Action.PICKED) {
                const doc = data[google.picker.Response.DOCUMENTS][0];
                const fileId = doc[google.picker.Document.ID];
                const fileName = doc[google.picker.Document.NAME] || "";
                onFilePicked(fileId, fileName);
              }
            });
          // Uncomment if Picker throws "developer key invalid":
          // if (GOOGLE_API_KEY) builder.setDeveloperKey(GOOGLE_API_KEY);
          // if (GOOGLE_APP_ID) builder.setAppId(GOOGLE_APP_ID);
          builder.build().setVisible(true);
        });
      }

    Note: gapi is a CDN global (apis.google.com/js/api.js already in index.html); no import
    needed. pickerLoaded promise caches the loaded state so repeated button taps do not re-load.
    TOKEN is the module-private variable at line 15 — Picker stays inside drive.js so TOKEN
    never leaks to shopping.js (resolves Open Question 2 from RESEARCH.md).

    The callback passes both fileId AND fileName so the caller can validate the file name
    before storing (mitigates T-02-01 spoofing threat).

<threat_model>
## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| Google Picker callback → listMeta kv | The picked file's id and name come from Google's Picker response; name must be validated before trusting the fileId |
| listMeta.fileId → drive.readFile | Stored fileId used for all subsequent Drive reads; if wrong file was picked, wrong data is merged |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-02-01 | Spoofing | Picker callback — wrong file picked | mitigate | Validate fileName.includes("einkaufsliste") in the Picker callback before calling kvSet; warn and abort if not matching |
| T-02-02 | Elevation of Privilege | TOKEN leaked from drive.js to shopping.js | mitigate | Picker is fully encapsulated in drive.js via openPickerForFile(); TOKEN is never exported; confirmed by grep gate in acceptance_criteria |
| T-02-03 | Tampering | Partner writes corrupted JSON to einkaufsliste.json | accept | mergeList() and decideSync() handle malformed remote.items gracefully (mergeList returns [] if items is not iterable); runtime errors caught in listSync.js catch block |
| T-02-04 | Denial of Service | gapi.load('picker') called before gapi is available (race on page load) | mitigate | Picker button only triggers on explicit user click (not on load); gapi loaded via CDN synchronously in index.html before any module runs |
| T-02-SC | Tampering | npm/pip/cargo installs | accept | No new packages — CDN Picker API is the same gapi script already in index.html |
</threat_model>
